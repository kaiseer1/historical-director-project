# Historical Director - momentum modifiers
#
# Applied by the amplified grant_claim action (see src/director/momentum.js).
# A pressed claim is a reason to go to war; these are the disposition to act on
# it. The Director never starts a war itself - it makes one affordable and
# attractive, and the ruler decides through ordinary game mechanics.
#
# Modelled on guiscard_modifier in 00_bookmark_modifiers.txt, which is the base
# game's own way of making a historically aggressive ruler behave like one:
#
#   guiscard_modifier = {
#       icon = flame_positive
#       army_maintenance_mult = -0.25
#       enemy_hostile_scheme_success_chance_add = -25
#       ai_war_cooldown = -1
#       ai_war_chance = 10
#   }
#
# These are stronger than that on purpose - roughly five times its war appetite
# - and every value below is still inside the range vanilla uses somewhere:
# ai_war_chance reaches 100, advantage reaches 6, levy_size reaches 0.5. They
# are applied for a fixed 30 years and expire on their own, so a mistaken
# approval is survivable.
#
# Only four modifier keys are used, and each was checked against real
# definitions in the base game rather than assumed:
#
#   ai_war_chance    willingness to declare war at all
#   ai_war_cooldown  how long the AI waits between wars (-1 removes it)
#   advantage        raw combat advantage in battle
#   levy_size        how many troops the levy raises
#
# ai_boldness and ai_zeal were considered and dropped: neither appears in any
# vanilla modifier definition, so their behaviour here would be a guess.
#
# To tune these, change the numbers here AND the constants at the top of
# src/director/momentum.js, which is what tells the player what they are
# approving. The two are not read from one another.

# --- reconquista -----------------------------------------------------------
# The recovery of lost ground. Weighted towards fielding and winning an army.
hd_reconquista_momentum = {
	icon = martial_positive
	ai_war_chance = 50
	ai_war_cooldown = -1
	advantage = 5
	levy_size = 0.25
}

# --- holy war --------------------------------------------------------------
# A war across a religious border, which the toolkit refuses to attach to a
# co-religionist. Weighted the same way; the difference in game terms is the
# piety that pays for the war, which the action grants separately.
hd_holy_war_momentum = {
	icon = piety_positive
	ai_war_chance = 50
	ai_war_cooldown = -1
	advantage = 5
	levy_size = 0.25
}

# --- succession pressure ---------------------------------------------------
# A disputed inheritance pressed by force. Slightly less eager than an outright
# war of conquest, and no combat bonus: the case is political, and what it buys
# is the readiness to escalate rather than an edge on the field.
hd_succession_momentum = {
	icon = flame_positive
	ai_war_chance = 40
	ai_war_cooldown = -1
	levy_size = 0.25
}

# than shoving.

hd_iberian_pressure_fervent = {

}

# --- Historical moments ----------------------------------------------------
# One per offered moment in src/director/moments.js. Between the momentum
# modifiers and the Iberian pressure ones in strength, and for a reason: a
# moment licenses a specific war against a specific realm that the record says
# happened, so it is more pointed than ambient regional pressure and less
# lavish than momentum, which pays for a war the player picked out by hand.
#
# Twenty-five years, because these are processes rather than campaigns: the
# Castilian-Leonese union took seventy years of dynastic accident, and the
# Almohad collapse ran three decades past Las Navas. A modifier that expires in
# ten would be describing a different kind of event.

hd_moment_union = {
	icon = martial_positive
	ai_war_chance = 25
	ai_war_cooldown = -0.5
	advantage = 3
	levy_size = 0.15
	monthly_prestige = 1
}

hd_moment_decline = {
	icon = martial_positive

	ai_war_cooldown = -0.5
	advantage = 3
	levy_size = 0.15
	monthly_piety = 1
}

# --- The Almohad collapse --------------------------------------------------
#
# Applied by the almohad_collapse macro action; see
# src/director/almohadCollapse.js for the severity table these mirror.
#
# The historical shape, briefly. Las Navas de Tolosa in 1212 did not destroy
# the Almohad state, it destroyed its authority in the peninsula. What followed
# was not conquest from outside but disintegration from within: Ibn Hud rose in
# Murcia in 1228, Ibn al-Ahmar founded the Nasrid emirate at Granada in 1238,
# and the third taifas emerged from provinces whose governors simply stopped
# obeying. The Christian crowns then took the pieces - Cordoba 1236, Valencia
# 1238, Seville 1248 - against opponents who were no longer one opponent.
#
# So the modelling puts the weight on *vassals leaving*, not on armies losing.
# `vassal_opinion` is the lever that does that, because vanilla's independence
# faction reads liege opinion directly: 00_factions.txt scores joining with
# OPINION_MULTIPLIER = -0.4 against a base reluctance of -150, so every point
# of vassal opinion lost here is 0.4 of a point toward a faction the caliph did
# not have yesterday. The military terms are secondary and deliberately mild -
# a caliph who cannot hold his emirs is the historical fact; a caliph who
# cannot win a battle is a different and smaller claim.
#
# Every key below appears in a vanilla modifier definition; none is assumed.
# vassal_opinion (111 uses), levy_size (152), monthly_income_mult (42),
# men_at_arms_maintenance (49), short_reign_duration_mult (16),
# monthly_prestige (275), advantage (64), garrison_size (67),
# knight_effectiveness_mult (48), ai_war_chance (17).
#
# Durations are set by the action rather than here, and run 20-30 years: the
# collapse took from 1212 to 1248, and a modifier that expires in ten would be
# describing a bad decade rather than the end of a caliphate.

# The authority slipping. Emirs are restive, the treasury is thinner, but the
# state still stands and could still recover.
hd_almohad_fraying = {
	icon = martial_negative
	vassal_opinion = -15
	levy_size = -0.10
	monthly_income_mult = -0.10
	short_reign_duration_mult = 0.25
}

# The historical 1220s. Provinces are being lost to their own governors rather
# than to invaders, and holding the army together now costs more than it earns.
hd_almohad_breaking = {

# --- Those taking the ground -----------------------------------------------
#
# Applied to the claimants named in the same proposal, Christian crown or
# breakaway emirate alike - the record allows both, and 1228-1248 had both.
#
# Deliberately modest next to the collapse modifiers above. The Director's line
# throughout is that it sets the stage and the ruler decides: a claimant who is
# handed a decisive military edge is not deciding anything. What this grants is
# the confidence to move on a neighbour who is visibly coming apart, which is
# what actually happened, and not the means to win the war that follows.
hd_taifa_ascendant = {
	icon = martial_positive
	ai_war_chance = 20
	monthly_prestige = 1
	advantage = 2
	knight_effectiveness_mult = 0.10
}

# Granted to a vassal who breaks away in the collapse - the Ibn Hud modifier,
# in effect. Small, and short: a new emirate's advantage was that nobody could
# spare the strength to reconquer it, not that it was strong.
hd_taifa_risen = {
	icon = county_modifier_opinion_positive
	monthly_prestige = 0.5
	garrison_size = 0.15
	levy_size = 0.10
}

# --- Historical wars --------------------------------------------------------
#
# Applied by the historical_war action (src/director/historicalWars.js) when a
# curated war starts. The war is real and CK3 fights it; these tilt it toward
# the outcome the record gives, and expire on their own. By kind rather than by
# war, so the next curated crusade needs a table entry, not a new modifier.
#
# Cheap by construction: a timed modifier is a few numbers on one character,
# read when stats are computed and gone when it expires. Paradox's own scripted
# historical war (the Raiktor bookmark) tilts its war with spawn_army, which
# puts real units on the map; this is lighter than that.
#
# Every key already appears in a vanilla modifier definition.

# The attacker in a war the record calls a crusade: the indulgences, the fleet
# that could be paid for, the nobles who came because the Pope said so.
# ai_war_chance matters only if start_war was refused and the claim is left to
# be pressed the ordinary way - during the war itself it does nothing.
hd_crusade_zeal = {
	icon = martial_positive
	advantage = 5
	levy_size = 0.20
	knight_effectiveness_mult = 0.15
	monthly_piety = 1
	ai_war_chance = 30
}

# The defender of ground the record says was lost: isolated, unsupported, and

hd_beleaguered_realm = {

	garrison_size = -0.25
	advantage = -2
}
