# Historical Director - Constrained action toolkit
#
# These are the only operations the model can cause to happen. The orchestrator
# never writes free-form script into the run file; it writes calls to the
# effects below with validated parameters. The engine-side vocabulary is
# therefore fixed at mod-install time, not at prompt time.
#
# Every effect ends by logging an HD:/;/applied line so the orchestrator can
# confirm execution actually happened rather than assuming it.

# --- spawn_character -------------------------------------------------------
# Introduce a historical figure. Spawns into the given title holder's court so
# the character exists somewhere sensible rather than floating in the void.
hd_spawn_character = {
	if = {
		limit = { exists = title:$COURT_TITLE$ exists = title:$COURT_TITLE$.holder }
		title:$COURT_TITLE$.holder = {
			save_scope_as = hd_host
			create_character = {
				employer = scope:hd_host
				name = "$NAME$"
				sex = $SEX$
				age = $AGE$
				culture = scope:hd_host.culture
				faith = scope:hd_host.faith
				dynasty = generate
				save_scope_as = hd_spawned
			}
		}
		debug_log = "HD:/;/applied/;/$TOKEN$/;/spawn_character/;/[scope:hd_spawned.GetID]"
	}
}

# --- set_relations ---------------------------------------------------------
# Enforce historically-attested hostility or amity between two living rulers.
hd_set_relations = {
	if = {
		limit = { exists = character:$ACTOR$ exists = character:$TARGET$ }
		character:$ACTOR$ = {
			add_opinion = {
				target = character:$TARGET$
				modifier = hd_historical_opinion
				opinion = $VALUE$
			}
		}
		debug_log = "HD:/;/applied/;/$TOKEN$/;/set_relations/;/$ACTOR$"
	}
}

# --- grant_claim -----------------------------------------------------------
# Seed a historically-grounded expansion by giving a ruler a pressable claim.
hd_grant_claim = {
	if = {
		limit = { exists = character:$ACTOR$ exists = title:$TITLE$ }
		character:$ACTOR$ = {
			add_pressed_claim = title:$TITLE$
		}
		debug_log = "HD:/;/applied/;/$TOKEN$/;/grant_claim/;/$ACTOR$"
	}
}

# --- adjust_title_tier -----------------------------------------------------
# Normalise a polity that has drifted above its historical tier: strip the
# over-ranked primary title and leave the ruler on the lower one they hold.
# Destructive, so it is guarded hard and always requires player approval.
hd_adjust_title_tier = {
	if = {
		limit = {
			exists = character:$ACTOR$
			exists = title:$TITLE$
			title:$TITLE$.holder ?= character:$ACTOR$
		}
		character:$ACTOR$ = {
			destroy_title = title:$TITLE$
		}
		debug_log = "HD:/;/applied/;/$TOKEN$/;/adjust_title_tier/;/$ACTOR$"
	}
}

# --- trigger_event ---------------------------------------------------------
# Narrative beat: fire one of the Director's own grounded events at a ruler.
hd_trigger_event = {
	if = {
		limit = { exists = character:$ACTOR$ }
		character:$ACTOR$ = {
			trigger_event = $EVENT$
		}
		debug_log = "HD:/;/applied/;/$TOKEN$/;/trigger_event/;/$ACTOR$"
	}
}
