# Historical Director - the two effects the mod still owns
#
# Everything else that used to live here is gone, and the reason is worth
# recording. Perception and the whole action toolkit were originally scripted
# effects called with parameters - hd_snapshot_region = { REGION = ... },
# hd_grant_claim = { ACTOR = ... } - and that does not survive contact with the
# engine: CK3 mangles $PARAM$ substitution inside a quoted debug_log string, so
# every record carrying one came back unparseable and snapshots never arrived.
#
# The orchestrator composes that script itself now (src/bridge/ck3Script.js),
# with the values already literal by the time CK3 sees them. The old effects sat
# here unused for several versions afterwards, which cost three things: an
# error in error.log on every load ("Variable 'hd_counties' is used but is never
# set", because the setter was in an effect nothing called), a documented wire
# format that had silently stopped being accurate, and a mod that read as
# though it implemented a toolkit it had not implemented for months.
#
# The upside of keeping the mod down to these two primitives is real and has
# been collected more than once: v0.3 widened the sphere, added a realm-footprint
# probe and changed the snapshot sweep, and needed no mod change and no game
# restart at all.
#
# Wire format is documented where it is produced, in src/bridge/ck3Script.js and
# src/bridge/protocol.js.

# --- Heartbeat -------------------------------------------------------------
# Cheap date tick, so the orchestrator can track in-game time without paying for
# a full snapshot on every pass. Called from hd_yearly_on_action and from the
# startup event.
hd_heartbeat = {
	debug_log = "HD:/;/date/;/[GetCurrentDate.GetStringShort]/;/[GetCurrentDate.GetDateAsTotalDays]"
}

# --- Pump liveness ---------------------------------------------------------
# Called at the top of every batch the orchestrator stages. The yearly watchdog
# in hd_on_actions.txt reads this flag to decide whether the execution pump
# still exists, so the mark has to come from script the pump actually ran, not
# from anything the game does on its own.
hd_mark_alive = {
	set_global_variable = { name = hd_pump_alive value = 1 }
}
