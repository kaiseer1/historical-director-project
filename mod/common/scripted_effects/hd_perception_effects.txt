# Historical Director - Perception layer
#
# Emits a filtered world-state snapshot into logs/debug.log over the same
# debug_log channel VOTC uses, but under the distinct "HD:" prefix so that
# both mods can run side by side without either parser seeing the other's data.
#
# Wire format (fields separated by /;/):
#   HD:/;/snapshot_begin/;/<token>/;/<date>/;/<totalDays>/;/<playerId>
#   HD:/;/realm/;/<id>/;/<ruler>/;/<primaryTitle>/;/<tier>/;/<countiesInSphere>/;/
#      <culture>/;/<faith>/;/<capital>/;/<dynasty>/;/<house>/;/<independent>/;/<government>
#   HD:/;/snapshot_end/;/<token>/;/<realmCount>
#   HD:/;/date/;/<date>/;/<totalDays>
#
# A snapshot is: one hd_snapshot_begin, one or more hd_snapshot_region (the
# sphere of influence), then one hd_snapshot_end. Realms are de-duplicated
# across regions, so overlapping regions are safe to pass.

hd_snapshot_begin = {
	clear_global_variable_list = hd_realm_set
	debug_log = "HD:/;/snapshot_begin/;/$TOKEN$/;/[GetCurrentDate.GetStringShort]/;/[GetCurrentDate.GetDateAsTotalDays]/;/[GetPlayer.GetID]"
}

# Walk every landed county in the region up to its top liege, collecting the
# distinct set of realms and, in the same pass, how many counties each of them
# holds inside the sphere. Regional footprint is the figure the Director wants
# to compare against history, so counting it here costs nothing extra.
hd_snapshot_region = {
	every_county_in_region = {
		region = $REGION$
		limit = { exists = holder }
		holder = {
			top_liege = {
				if = {
					limit = {
						NOT = { is_target_in_global_variable_list = { name = hd_realm_set target = this } }
					}
					add_to_global_variable_list = { name = hd_realm_set target = this }
					set_variable = { name = hd_counties value = 0 }
				}
				change_variable = { name = hd_counties add = 1 }
			}
		}
	}
}

hd_log_realm = {
	debug_log = "HD:/;/realm/;/[THIS.Char.GetID]/;/[THIS.Char.GetTitledFirstNameNicknamedNoTooltipRegnal]/;/[THIS.Char.GetPrimaryTitle.GetNameNoTooltip]/;/[THIS.Char.GetPrimaryTitle.GetRankConcept]/;/[THIS.Var('hd_counties').GetValue]/;/[THIS.Char.GetCulture.GetName]/;/[THIS.Char.GetFaith.GetName]/;/[THIS.Char.GetCapitalLocation.GetNameNoTooltip]/;/[THIS.Char.GetDynasty.GetName]/;/[THIS.Char.GetHouse.GetName]/;/[THIS.Char.IsIndependentRuler]/;/[THIS.Char.GetGovernment.GetNameNoTooltip]"
	remove_variable = hd_counties
}

hd_snapshot_end = {
	every_in_global_list = {
		variable = hd_realm_set
		hd_log_realm = yes
	}
	debug_log = "HD:/;/snapshot_end/;/$TOKEN$"
	clear_global_variable_list = hd_realm_set
}

# Cheap date heartbeat, so the orchestrator can track in-game time without
# paying for a full snapshot on every tick.
hd_heartbeat = {
	debug_log = "HD:/;/date/;/[GetCurrentDate.GetStringShort]/;/[GetCurrentDate.GetDateAsTotalDays]"
}

# --- Locating the player ---------------------------------------------------
# The sphere of influence has to start somewhere, and the orchestrator cannot
# work out where the player sits from county names alone. Rather than snapshot
# the world to find out, we ask the engine directly: test the player's capital
# against every region the base game defines, and log the hits.
#
# Two things here were learned the hard way. The tests run inside every_player
# rather than trusting the run file's root scope, because `capital_county`
# needs a character scope and the ?= makes a missing one fail silently - a
# player sitting in Rayy reported no region at all, despite Rayy being squarely
# inside world_persia. And it probes every region rather than a curated few,
# because a curated list is a promise to keep curating it and each omission
# costs a restart to discover. Sixty-odd triggers is nothing; the orchestrator
# decides which of the hits it is willing to reason about.

hd_probe_region = {
	if = {
		limit = {
			capital_county.title_province ?= { geographical_region = $REGION$ }
		}
		debug_log = "HD:/;/in_region/;/$REGION$"
	}
}

# --- Pump liveness ------------------------------------------------------
# Called at the end of every batch the orchestrator stages. The yearly
# watchdog in hd_on_actions.txt reads this flag to decide whether the pump
# still exists, so the mark has to come from script the pump actually ran,
# not from anything the game does on its own.
hd_mark_alive = {
	set_global_variable = { name = hd_pump_alive value = 1 }
}
